    37	        if (event.targetGisuId() == null) {

    39	        }

    42	        if (challengers.isEmpty()) {

    55	            getChapterUseCase.getChapterMapByGisuIdsAndSchoolIds(Set.of(event.targetGisuId()), schoolIds);

    68	                continue;
    69	            }

    71	            if (matches(event, challenger.gisuId(), chapter.id(), schoolId, challenger.part())) {
    72	                targetMemberIds.add(challenger.memberId());
    73	            }
    74	        }

    89	    public void relay() {
    90	        List<EventOutbox> outboxes = claimPublishable();
    91	        for (EventOutbox outbox : outboxes) {
    92	            relayOne(outbox);
    93	        }

    96	    private List<EventOutbox> claimPublishable() {
    97	        List<EventOutbox> outboxes = transactionTemplate.execute(status -> {
    98	            Instant now = Instant.now();
    99	            List<EventOutbox> publishableOutboxes = loadEventOutboxPort.listPublishable(batchSize, now);
   100	            Instant leaseUntil = now.plus(PROCESSING_LEASE);
   101	            publishableOutboxes.forEach(outbox -> outbox.markProcessing(leaseUntil));
   102	            saveEventOutboxPort.saveAll(publishableOutboxes);
   103	            return publishableOutboxes;
   104	        });
   105	        return outboxes == null ? List.of() : outboxes;
   106	    }
   107	
   108	    private void relayOne(EventOutbox outbox) {
   109	        try {
   110	            publish(outbox);
   111	        } catch (RuntimeException e) {
   112	            recordFailure(outbox, e);
   113	        }
   114	    }
   115	
   116	    private void publish(EventOutbox outbox) {
   117	        // 발행 시점에 저장해둔 traceparent로 원 요청 trace를 복원해 span link로 연결한다.
   118	        // relay span은 폴러의 task span 아래 자식으로 생성되며, link로 원 요청 trace와 이어진다.
   119	        TraceContext origin = W3CTraceparent.restore(tracer, outbox.getTraceparent());
   120	        Span span = newRelaySpan(origin);
   121	        try (Tracer.SpanInScope ignored = tracer.withSpan(span)) {
   122	            doPublish(outbox);
   123	        } catch (RuntimeException e) {
   124	            span.error(e);
   125	            throw e;
   126	        } finally {
   127	            span.end();
   128	        }
   129	    }
   130	
   131	    private Span newRelaySpan(TraceContext origin) {
   132	        Span.Builder builder = tracer.spanBuilder().name(RELAY_SPAN_NAME);
   133	        if (origin != null) {
   134	            builder.addLink(new Link(origin));
   135	        }
   136	        return builder.start();
   137	    }
   138	
   139	    private void doPublish(EventOutbox outbox) {
   140	        // 이벤트 발행과 published 상태 변경을 하나의 트랜잭션으로 묶는다.
   141	        // 리스너는 @TransactionalEventListener(AFTER_COMMIT)라 markPublished가 커밋된 뒤에야 부작용이 발동하므로,
   142	        // "발행은 됐는데 published 처리는 실패"하는 중복 윈도우가 사라진다.
   143	        transactionTemplate.executeWithoutResult(status -> {
   144	            DomainEvent event = deserializer.deserialize(outbox);
   145	            eventPublisher.publishEvent(event);
   146	            outbox.markPublished();
   147	            saveEventOutboxPort.save(outbox);
   148	        });
   149	    }
   150	
   151	    private void recordFailure(EventOutbox outbox, RuntimeException e) {
   152	        transactionTemplate.executeWithoutResult(status -> {
   153	            outbox.recordFailure(errorMessage(e), nextAttemptAt(outbox), maxAttempts);
   154	            saveEventOutboxPort.save(outbox);
   155	        });
   156	    }
   157	
   158	    private Instant nextAttemptAt(EventOutbox outbox) {
   159	        long multiplier = 1L << Math.min(outbox.getAttempts(), 6);
   160	        Duration backoff = BASE_BACKOFF.multipliedBy(multiplier);
   161	        if (backoff.compareTo(MAX_BACKOFF) > 0) {
   162	            backoff = MAX_BACKOFF;
   163	        }
   164	        return Instant.now().plus(backoff);
   165	    }

   168	        if (e.getMessage() == null || e.getMessage().isBlank()) {
   169	            return e.getClass().getName();
   170	        }
   171	        return e.getMessage();
   172	    }
